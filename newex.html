function updateChart() { if (!fullHistory.length) return; const filteredHistory = filterHistoryByRange(currentRange); if (filteredHistory.length === 0) return; // Подготавливаем данные для графика С НАЦЕНКОЙ const times = filteredHistory.map(item => item.time); const ratesWithMarkup = filteredHistory.map(item => item.rate + MARKUP_RUB_USDT); // Обновляем статистику (тоже с наценкой) const filteredForStats = filteredHistory.map(item => ({ rate: item.rate + MARKUP_RUB_USDT })); const stats = calculateStats(filteredForStats); if (stats) updateStats(filteredForStats); if (chart) { chart.data.labels = times; chart.data.datasets[0].data = ratesWithMarkup; chart.update(); } else { const ctx = document.getElementById('rateChart').getContext('2d'); chart = new Chart(ctx, { type: 'line', data: { labels: times, datasets: [{ label: `Курс RUB за 1 USDT (с наценкой ${MARKUP_RUB_USDT} руб)`, data: ratesWithMarkup, borderColor: '#10b981', backgroundColor: 'rgba(16, 185, 129, 0.1)', borderWidth: 2, pointRadius: 2, pointBackgroundColor: '#6ee7b7', pointBorderColor: '#0b3d3d', pointHoverRadius: 5, fill: true, tension: 0.3 }] }, options: { responsive: true, maintainAspectRatio: true, interaction: { mode: 'index', intersect: false, }, plugins: { legend: { labels: { color: '#99f6e4', font: { size: 12 } } }, tooltip: { callbacks: { label: function(context) { return `${context.parsed.y.toFixed(6)} RUB за 1 USDT`; } } } }, scales: { y: { grid: { color: 'rgba(110, 231, 183, 0.1)' }, ticks: { color: '#99f6e4' } }, x: { grid: { color: 'rgba(110, 231, 183, 0.1)' }, ticks: { color: '#99f6e4', maxRotation: 45, autoSkip: true, maxTicksLimit: 10 } } } } }); } }
